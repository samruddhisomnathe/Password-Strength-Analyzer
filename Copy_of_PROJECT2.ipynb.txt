        "import re\n",
        "\n",
        "def analyze_password(password):\n",
        "    score = 0\n",
        "\n",
        "    # Length Check\n",
        "    if len(password) >= 8:\n",
        "        score += 1\n",
        "\n",
        "    # Uppercase Check\n",
        "    if re.search(r\"[A-Z]\", password):\n",
        "        score += 1\n",
        "\n",
        "    # Lowercase Check\n",
        "    if re.search(r\"[a-z]\", password):\n",
        "        score += 1\n",
        "\n",
        "    # Digit Check\n",
        "    if re.search(r\"\\d\", password):\n",
        "        score += 1\n",
        "\n",
        "    # Special Character Check\n",
        "    if re.search(r\"[!@#$%^&*(),.?\\\":{}|<>]\", password):\n",
        "        score += 1\n",
        "\n",
        "    # Strength Evaluation\n",
        "    if score <= 2:\n",
        "        strength = \"Weak\"\n",
        "    elif score == 3 or score == 4:\n",
        "        strength = \"Medium\"\n",
        "    else:\n",
        "        strength = \"Strong\"\n",
        "\n",
        "    print(\"\\nPassword Analysis\")\n",
        "    print(\"------------------\")\n",
        "    print(\"Password Strength:\", strength)\n",
        "    print(\"Score:\", score, \"/5\")\n",
        "\n",
        "    if strength == \"Weak\":\n",
        "        print(\"Suggestion: Use uppercase, lowercase, numbers, and special characters.\")\n",
        "    elif strength == \"Medium\":\n",
        "        print(\"Suggestion: Add more complexity to make it stronger.\")\n",
        "    else:\n",
        "        print(\"Excellent! Your password is strong.\")\n",
        "\n",
        "# Main Program\n",
